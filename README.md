# Fetches and merges changes from the current branch's default upstream branch.
git pull
git pull origin <branch name>

# this will tell you about the branches you have
git branch 

# this command to switch branch
git checkout <branch name>
git checkout -b <branch name> # create a branch and switch to the branch simultaneously

# Stage changes
git add .  # add all
git add <filename> # add certain file

# commit changes
git commit -m "write a long message here"

# push to github
git push origin <branch-name>
